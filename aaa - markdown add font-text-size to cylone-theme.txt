function escapeHTML(str) {
    return str.replace(/[&<>'"]/g, tag => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;'
    }[tag] || tag));
}

function print(html) {
    const term = document.getElementById('video-layer');
    if (!term) return;

    // 1. Split incoming stream by ANSI escape sequences
    const parts = String(html).split(/(\x1B\[[0-9;]*[a-zA-Z])/);
    
    let currentHtml = "";
    let spanOpen = false;
    let clearPrevious = 0;
    
    for (let i = 0; i < parts.length; i++) {
        let part = parts[i];
        if (!part) continue;
        
        if (part.startsWith('\x1B[')) {
            let code = part.substring(2);
            let cmd = code.slice(-1);
            let params = code.slice(0, -1).split(';');
            
            if (cmd === 'A') { // VT100 Cursor Up
                clearPrevious += (parseInt(params[0], 10) || 1);
            } else if (cmd === 'm') { // VT100 Graphic Rendition (Colors)
                if (params[0] === '0' || params[0] === '') {
                    if (spanOpen) { currentHtml += '</span>'; spanOpen = false; }
                } else if (params[0] === '30') {
                    if (spanOpen) { currentHtml += '</span>'; }
                    currentHtml += `<span style="color: #444;">`; // dark gray for visibility on black
                    spanOpen = true;
                } else if (params[0] === '38' && params[1] === '5') {
                    let colorCode = parseInt(params[2], 10);
                    let hex = 'var(--fg-color, #0f0)';
                    if (colorCode === 22) hex = '#005f00';
                    else if (colorCode === 28) hex = '#008700';
                    else if (colorCode === 46) hex = '#00ff00';
                    
                    if (spanOpen) { currentHtml += '</span>'; }
                    currentHtml += `<span style="color: ${hex};">`;
                    spanOpen = true;
                }
            }
        } else {
            currentHtml += part;
        }
    }
    
    if (spanOpen) currentHtml += '</span>';
    
    // 2. Erase previous lines if Cursor Up was used
    if (clearPrevious > 0) {
        let divs = Array.from(term.querySelectorAll('div:not(.term-spacer)'));
        for (let j = 0; j < clearPrevious && divs.length > 0; j++) {
            let lastDiv = divs.pop();
            if (lastDiv && lastDiv.parentNode) lastDiv.parentNode.removeChild(lastDiv);
        }
    }
    
    // 3. Line-by-Line DOM Generation (Perserving inline tags across lines)
    if (currentHtml) {
        let normalized = currentHtml.replace(/\n/g, '<br>');
        let rawLines = normalized.split('<br>');
        
        // Remove trailing empty line caused by trailing \n or <br> to keep spacing tight
        if (rawLines.length > 0 && rawLines[rawLines.length - 1] === "") {
            rawLines.pop(); 
        }
        
        let activeTags = [];
        for (let line of rawLines) {
            let prefix = activeTags.join('');
            
            // Extract open tags so we can re-apply them to the next line
            let tagRegex = /<\/?([a-zA-Z0-9]+)[^>]*>/g;
            let match;
            while ((match = tagRegex.exec(line)) !== null) {
                let fullTag = match[0];
                let tagName = match[1].toLowerCase();
                if (fullTag.startsWith('</')) {
                    activeTags.pop();
                } else if (!fullTag.endsWith('/>') && !['br','img','hr','input'].includes(tagName)) {
                    activeTags.push(fullTag);
                }
            }
            
            // Seal this line safely
            let suffix = activeTags.slice().reverse().map(tag => {
                let name = tag.match(/<([a-zA-Z0-9]+)/)[1];
                return `</${name}>`;
            }).join('');
            
            let div = document.createElement('div');
            let finalHtml = prefix + line + suffix;
            div.innerHTML = finalHtml === "" ? "&nbsp;" : finalHtml;
            term.appendChild(div);
        }
        
        term.scrollTop = term.scrollHeight; // Auto-scroll to bottom
    }
}

// Special print for DOS output with robust ANSI-style [.0] - [.9] state tracking
function printDOS(text) {
    let activeFg = null;
    let activeBg = null;
    let spanOpen = false;

    let formatted = escapeHTML(String(text == null ? '' : text))
        // Process color codes sequentially to prevent messy nesting
        .replace(/\[\.([0-9]?)\]/g, function(match, code) {
            // [.] Reset everything
            if (code === '') {
                activeFg = null;
                activeBg = null;
                let res = spanOpen ? '</span>' : '';
                spanOpen = false;
                return res;
            }
            
            let num = parseInt(code, 10);
            if (num === 0) activeFg = `var(--fg-color)`;
            else if (num >= 1 && num <= 4) activeFg = `var(--text-fg${num})`;

            if (num === 5) activeBg = `var(--bg-color)`;
            else if (num >= 6 && num <= 9) activeBg = `var(--text-bg${num})`;
            
            let res = spanOpen ? '</span>' : '';
            spanOpen = true;
            let style = '';
            if (activeFg) style += `color: ${activeFg}; `;
            if (activeBg) style += `background-color: ${activeBg}; `;
            return res + `<span style="${style}">`;
        });

    if (spanOpen) formatted += '</span>';

    // Apply standard spacing rules
    formatted = formatted
        .replace(/ {2,}/g, run => ' ' + new Array(run.length).join('&nbsp;'))
        .replace(/\r?\n/g, '<br>');
        
    print(formatted);
}

function updatePrompt() {
    if (window.CylonDOS) {
        let path = window.CylonDOS.pwd();
        
        // Ensure path always has a trailing slash for formatting (e.g. ram:/docs/)
        if (path !== '(no drive)' && !path.endsWith('/')) { path += '/'; }
        sysPrompt = path;
        
        var promptEl = document.getElementById('kb-prompt');
        
        // BUGFIX: Allow app-cli mode to use the DOS prompt overlay
        if (promptEl && (osMode === 'terminal' || osMode === 'app-cli')) {
            promptEl.textContent = sysPrompt;
        }
        
        // Prompt keyboard renderer to update dimensions against the new prompt length
        if (typeof window.renderInputLine === 'function') {
            window.renderInputLine();
        }
    }
}

window.updatePrompt = updatePrompt; // Expose globally for the virtual keyboard

// ==== STATE & VIRTUAL KEY DISPATCH ====
window.setOSMode = function(mode, promptText) {
    osMode = mode;
    inputBuffer = "";
    cursorPos = 0;
    selAnchor = null;

    var promptEl = document.getElementById('kb-prompt');
    if (!promptEl) return;

    if (mode === 'terminal' || mode === 'app-cli') {
        if (window.updatePrompt) window.updatePrompt();
    } else if (mode === 'app') {
        var activeFrame = document.querySelector('.page-frame.active');
        var title = activeFrame ? activeFrame.dataset.title : 'App';
        promptEl.textContent = `[ ${title} ]`;
    } else if (mode === 'app-edit' || mode === 'app-textarea') {
        if (promptText) promptEl.textContent = promptText;
    }

    if (window.renderInputLine) window.renderInputLine();
};

// Sync OS Mode dynamically based on what the user's browser is focused on
function syncAppMode() {
    if (osMode !== 'app' && osMode !== 'app-cli') return;
    var activeFrame = document.querySelector('.page-frame.active');
    
    // If the active element is the app iframe, switch to APP mode. Otherwise, CLI mode.
    if (activeFrame && document.activeElement === activeFrame) {
        if (osMode !== 'app') window.setOSMode('app');
    } else {
        if (osMode !== 'app-cli') window.setOSMode('app-cli');
    }
}

window.injectAndExecute = async function(cmdStr, opts) {
    if (osMode !== 'terminal' && osMode !== 'app') window.setOSMode('app'); 
    
    inputBuffer = cmdStr;
    cursorPos = inputBuffer.length;
    if (window.renderInputLine) window.renderInputLine();

    inputBuffer = "";

    selAnchor = null;

    if (window.renderInputLine) window.renderInputLine();
    return ok !== false;
};

            `${e.data.id}: `
        );

        inputBuffer = e.data.value || ""; 
        cursorPos = inputBuffer.length;
        
        window.showFullKeyboard();
        window.updateOverlayInput();

        if (e.data.rect) window.panToRect(e.data.rect);
    } else if (e.data.type === 'CYLON_PRINT') {
        if (typeof print === 'function') print(e.data.text);
    } else if (e.data.type === 'CYLON_POP_HTM') {
        popupActive = true;
        let shield = document.getElementById('qandy-popup-shield');
        if (shield) shield.style.display = 'block';
    } else if (e.data.type === 'CYLON_POP_HIDE') {
        popupActive = false;
        let shield = document.getElementById('qandy-popup-shield');
        if (shield) shield.style.display = 'none';
    } else if (e.data.type === 'QANDY_INPUT_RECT') {
        window.panToRect(e.data.rect);
    } else if (e.data.type === 'QANDY_SET_MODE') {
        window.setOSMode(e.data.mode);
        if (osMode === 'terminal') { window.showFullKeyboard(); }
    } else if (e.data.type === 'QANDY_PASTE') {
        if (window.pasteText) window.pasteText(e.data.text);
    } else if (e.data.type === 'QANDY_APP_CLICKED') {
        // If the user clicks the app's visual layer, abort the hybrid CLI mode

            window.setOSMode('app');
            if (typeof window.hideFullKeyboard === 'function') window.hideFullKeyboard();
        }
    }
});

window.addEventListener('copy', function(e) {
    // If the user highlighted standard text in the history terminal, let the browser copy it natively!
    if (window.getSelection().toString() !== "") return;
    
    // Otherwise, copy the selected text from our virtual DOS input buffer
    if (selAnchor !== null && selAnchor !== cursorPos) {
        let start = Math.min(selAnchor, cursorPos);
        let end = Math.max(selAnchor, cursorPos);
        let selected = inputBuffer.slice(start, end);
        if (selected) {
            e.clipboardData.setData('text/plain', selected);
            e.preventDefault(); // Tell browser we handled it
        }
    }
});

window.addEventListener('cut', function(e) {
    // Let native history selections copy freely
    if (window.getSelection().toString() !== "") return;

    if (selAnchor !== null && selAnchor !== cursorPos) {
        let start = Math.min(selAnchor, cursorPos);
        let end = Math.max(selAnchor, cursorPos);
        let selected = inputBuffer.slice(start, end);
        if (selected) {
            // 1. Write to OS Clipboard
            e.clipboardData.setData('text/plain', selected);
            e.preventDefault();
            
            // 2. Erase from virtual input buffer
            inputBuffer = inputBuffer.slice(0, start) + inputBuffer.slice(end);
            cursorPos = start;
            selAnchor = null;
            if (window.renderInputLine) window.renderInputLine();
        }
    }
});

window.addEventListener('paste', function(e) {
    let text = (e.clipboardData || window.clipboardData).getData('text');
    if (text && window.pasteText) {
        window.pasteText(text);
        e.preventDefault();
    }
});

// Boot is just the commands a user could type themselves.
async function bootMount() {
    var dos = window.CylonDOS;
    var run = function (cmd) { return window.injectAndExecute(cmd, { boot: true }); };
    
    var dbs = window.indexedDB && window.indexedDB.databases ? await window.indexedDB.databases() : [];
    var dbExists = dbs.some(function(db) { return db.name === 'CylonDOS'; });

    // 1. Installed: boot from the IndexedDB partition
    if (dbExists) {
        var parts = await dos.partitions();
        var boot = parts.filter(function (p) { return p.boot; })[0] ||
                   parts.filter(function (p) { return p.name === 'cylon'; })[0];

        if (boot && await run('mount ' + boot.name + ' idb')) {
            await run(boot.name + ':');
            return;
        }
    }

    // 2. Not installed: attempt to boot from the cylon.json image into RAM
    if (await run('mount cylon.json ram cylon')) {
        await run('cylon:');
        printDOS("[.2]System booted to RAM drive, run 'install' to setup indexDB[.]\n");
    } else {
        // 3. No image available: blank RAM drive
        await run('mount ram:');
        await run('ram:');
        printDOS("[.2]Booted to RAM drive (cylon.json not found).\nInput 'export cylon.json' to download your work later[.]\n");
    }
}

// Screen Wrapper dynamically scales the OS Coordinate Grid (Gfx Engine)
window.fitScreenToGrid = function() {
    if (!window.CylonGFX) return;
    const wrapper = document.getElementById('screen-wrapper');
    if (!wrapper) return;

    const screenW = window.innerWidth;
    const barH = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--bar-height')) || 60;
    let screenH = window.innerHeight - barH;

    const cols = window.CylonGFX.gridWide || 8;
    const rows = window.CylonGFX.gridTall || 4;

    let w = screenW, h = screenH;
    if (window.CylonGFX.gfxGridStretch) {
        const aspect = cols / rows;
        h = screenW / aspect;
        if (h > screenH) {
            h = screenH;
            w = screenH * aspect;
        }
        w = Math.floor(w * 0.98);
        h = Math.floor(h * 0.98);
    } else {
        w = cols * window.CylonGFX.gfxTileSize;
        h = rows * window.CylonGFX.gfxTileSize;
    }

    wrapper.style.width = w + 'px';
    wrapper.style.height = h + 'px';
};

    root.style.setProperty('--case-fg', vars.themeCaseFG);
    root.style.setProperty('--key-bg', vars.themeKeyUpBG);
    root.style.setProperty('--key-fg', vars.themeKeyUpFG);
    root.style.setProperty('--key-hl-bg', vars.themeKeyDownBG);
    root.style.setProperty('--key-hl-fg', vars.themeKeyDownFG);

    root.style.setProperty('--html-link', vars.htmlLink);
    root.style.setProperty('--html-visited', vars.htmlVisited);
    root.style.setProperty('--html-hover', vars.htmlHover);
    root.style.setProperty('--html-active', vars.htmlActive);

    for(let i=6; i<=9; i++) root.style.setProperty(`--text-bg${i}`, vars[`themeTextBg${i}`]);

    let sizeMap = {
        "0": "clamp(12px, 3vw, 16px)",
        "1": "clamp(14px, 3.5vw, 18px)",
        "2": "clamp(16px, 4.5vw, 22px)",
        "3": "clamp(18px, 5.5vw, 26px)",
        "4": "clamp(22px, 7vw, 32px)",
        "5": "clamp(28px, 9vw, 42px)"
    };
    root.style.setProperty('--term-font-size', sizeMap[vars.themeTextSize] || sizeMap["2"]);
    if (typeof window.renderInputLine === 'function') window.renderInputLine();

    document.querySelectorAll('iframe.page-frame').forEach(frame => {
        if (frame.contentWindow) {
            frame.contentWindow.postMessage({type: 'QANDY_UPDATE_COLORS', colors: window.CylonColors}, (window.location.origin && window.location.origin !== "null" ? window.location.origin : "*"));
        }
    });
};

window.desktopInit = async function() {
    let dos = window.CylonDOS;
    
    // 1. Load system colors from /a/ retro-objects BEFORE autoexec loads apps
    await window.applyThemeFromAlpha();

    // 2. Initialize Headless GFX Physics Engine for the OS Terminal
    if (window.CylonGFX) {
        let mnt = await dos.mounts();
        let bootDrive = mnt.find(m => m.current) || {name: 'ram'};
        
        let startSector = 'Aa';
        try {
            let loaded = await dos.load(bootDrive.name + ':/a/gfxSector');
            if (loaded && loaded.trim().length === 2) startSector = loaded.trim();
        } catch(e) {}
        
        // This runs safely in the background tracing coordinates even in pure terminal mode
        await window.CylonGFX.init(bootDrive.name + ':/w/' + startSector);
        
        // Connect the OS Terminal wrapper to the dynamic screen sizing
        window.fitScreenToGrid();
        window.addEventListener('resize', window.fitScreenToGrid);
    }

    // 3. Check for cylon.cfg (our autoexec file)
    let cfgText = "";
    if (await dos.exists('cylon.cfg')) { cfgText = await dos.load('cylon.cfg'); }
    
    // 4. Process the autoexec file line-by-line
    let lines = cfgText.split('\n');
    for (let line of lines) {
        line = line.trim();
        if (!line || line.startsWith('//') || line.startsWith(';')) continue;
        if (line.toLowerCase().endsWith('.htm') || line.toLowerCase().endsWith('.html')) {
            try { await window.PageOpen(line); } catch(e) {
                printDOS(`{.yellow}Autoexec: Could not open ${line}{.}\n`);
            }
        }
    }
};

function initializeDos() {
    if (dosInitialized) return Promise.resolve();
    if (dosInitializing) return dosInitializing;
    if (!window.CylonDOS) return Promise.reject(new Error('<font color=yellow>CylonDOS is not loaded!</font>'));

    dosInitializing = window.CylonDOS.ready().then(bootMount).then(function () {
        dosInitialized = true;
        updatePrompt();
        if (typeof window.desktopInit === 'function') window.desktopInit();
    });
    
    dosInitializing.catch(function () { dosInitializing = null; });
    return dosInitializing;
}

window.cylonDOSInitialize = initializeDos;

// Changed to async to natively support DOS operations seamlessly
async function command(cmdStr, opts) {
    cmdStr = cmdStr.trim();
    
    if (cmdStr) {
        cmdHistory.push(cmdStr);
        historyIdx = -1;
    }

    // Moved up to print the prompt and text BEFORE the empty check (this enables blank lines)
    // Removed trailing `<br><br>` because `print()` naturally wraps into block elements, saving blank vertical lines
    print(sysPrompt + escapeHTML(cmdStr)); 

    if (!cmdStr) return;

    // Handled locally since it directly manipulates the DOM
    if (cmdStr.toLowerCase() === 'cls' || cmdStr.toLowerCase() === 'clear') {
        document.getElementById('video-layer').innerHTML = '<div class="term-spacer"></div>';
        return;
    }

    // --- CYLON DOS COMMAND INTERCEPT ---
    if (window.CylonDOS && window.handleCylonCommand) {
        // Ensure DOS is loaded before dispatching (the bootloader runs *inside* init, so it skips this)
        if (!(opts && opts.boot)) await initializeDos();
        try {
            let result = await window.handleCylonCommand(cmdStr);
            updatePrompt();
            if (result && result.handled) {
                if (result.output) printDOS(result.output);
                return result.ok !== false; // Stop here, DOS handled the command (false = it failed).
            }
        } catch (error) {
            printDOS("Error: [.4]" + (error.message || String(error))+"[.]");
            return false; 
        }
    }

    // --- LEGACY APP / SYSTEM COMMANDS ---
    if (cmdStr.toLowerCase() === 'exit') {
        showTerminal();
        return;
    }

    let closeMatch = cmdStr.match(/^close\s+(\S+)$/i);
    if (closeMatch) {
        PageClose(closeMatch[1]);
        return;
    }

    if (/\.html?$/i.test(cmdStr)) {
      try {
          await PageOpen(cmdStr);
      } catch (error) {
          printDOS(error.message || String(error));
      }
      return;
    }

    print("Error: [.4]Unknown command: " + escapeHTML(cmdStr)+"[.]");
}

function navigateHistory(direction) {
    if (cmdHistory.length === 0) return;
    
    if (direction === -1) { // Up
        if (historyIdx < cmdHistory.length - 1) historyIdx++;
    } else { // Down
        if (historyIdx > 0) historyIdx--;
        else historyIdx = -1;
    }
    
    if (historyIdx === -1) {
        inputBuffer = "";
    } else {
        inputBuffer = cmdHistory[cmdHistory.length - 1 - historyIdx];
    }
}

function powerOn() {
    var overlay = document.createElement('div');
    overlay.id = 'qandy-power-overlay';
    Object.assign(overlay.style, {
      position: 'fixed', left: '0', top: '0', width: '100%', height: '100%',
      zIndex: '300', background: '#000000', display: 'flex', // Solid black effectively simulates screen off
      alignItems: 'center', justifyContent: 'center', userSelect: 'none'
    });

    var label = document.createElement('div');
    label.textContent = 'TOUCH, PRESS, OR CLICK ANY KEY';
    Object.assign(label.style, {
      color: '#fff', fontFamily: 'monospace', fontSize: '28px',
      letterSpacing: '2px', textAlign: 'center', textShadow: '0 1px 0 rgba(0,0,0,0.6)'
    });

    var helper = document.createElement('div');
    helper.textContent = 'TO POWER ON THE QANDY POCKET COMPUTER';
    Object.assign(helper.style, {
      color: '#ffffff', fontFamily: 'monospace', fontSize: '28px', marginTop: '8px'
    });

    var stack = document.createElement('div');
    stack.style.display = 'flex'; stack.style.flexDirection = 'column'; stack.style.alignItems = 'center';
    stack.appendChild(label); stack.appendChild(helper);
    overlay.appendChild(stack);
    document.body.appendChild(overlay);

    window.triggerOSMenu = function() {
        if (osMode === 'terminal') return; // Future expansion: terminal OS menu
        // Send the OS menu command down to the currently active app iframe
        const activeFrame = document.querySelector('.page-frame.active');
        if (activeFrame) {
            activeFrame.contentWindow.postMessage({ type: 'QANDY_OS_MENU' }, (window.location.origin && window.location.origin !== "null" ? window.location.origin : "*"));
        }
    };

    function engageHandler(e) {
      overlay.removeEventListener('click', engageHandler, true);
      document.removeEventListener('keydown', engageHandler, true);
      
      // Request Fullscreen mode to hide the browser UI (URL bar, tabs)
      try {
          if (document.documentElement.requestFullscreen) {
              document.documentElement.requestFullscreen().catch(err => console.warn(err));
          } else if (document.documentElement.webkitRequestFullscreen) { /* Safari */
              document.documentElement.webkitRequestFullscreen();
          }
      } catch (err) {
          console.warn('Fullscreen API not supported or blocked.');
      }

      try {
        if (typeof sound_js === 'function') { sound_js(); }
      } catch (e) { console.warn('qandy: error while initializing sound', e); }

      beep(950, 32, 1); 

      if (overlay.parentNode) overlay.parentNode.removeChild(overlay);
      loadNext();
    }    

    overlay.addEventListener('click', engageHandler, { capture: true, once: true });
    document.addEventListener('keydown', engageHandler, { capture: true, once: true });
}

};

        if (e.key === 'Shift') {
            window.setModifier('shift', 'physical-shift', true);
            window.setKeyVisual('lshift', 'active', true);
            window.setKeyVisual('rshift', 'active', true);
            window.updateKeyLabels();
        } else if (e.key === 'Control') {
            window.setModifier('ctrl', 'physical-ctrl', true);
            window.setKeyVisual('lctrl', 'active', true);
            window.setKeyVisual('rctrl', 'active', true);
            window.updateKeyLabels();
        } else if (e.key === 'Alt') {
            window.setModifier('alt', 'physical-alt', true);
            window.setKeyVisual('lalt', 'active', true);
            window.setKeyVisual('ralt', 'active', true);
            window.updateKeyLabels();
            e.preventDefault();
        } else if (e.key === 'CapsLock') {
            window.caps = e.getModifierState && e.getModifierState('CapsLock');
            window.setKeyVisual('caps', 'locked', window.caps);
            window.updateKeyLabels();
        } else if (e.key === 'Meta' || e.key === 'OS' || e.key === 'Windows') { 
            window.toggleFullKeyboard(); 
            e.preventDefault(); 
        } else if (e.key === 'Backspace') { window.press({id: 'back'}); e.preventDefault(); }
        else if (e.key === 'Enter') { window.press({id: 'enter'}); e.preventDefault(); }
        else if (e.key === 'ArrowUp') { window.press({id: 'up'}); e.preventDefault(); }
        else if (e.key === 'ArrowDown') { window.press({id: 'down'}); e.preventDefault(); }
        else if (e.key === 'ArrowLeft') { window.press({id: 'left', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === 'ArrowRight') { window.press({id: 'right', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === 'Home') { window.press({id: 'home', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === 'End') { window.press({id: 'end', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === 'Insert') { window.press({id: 'ins', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === 'Delete') { window.press({id: 'del', shift: e.shiftKey}); e.preventDefault(); }
        else if (e.key === '`' || e.key === '~') {
            e.preventDefault();
            //window.toggleFullKeyboard();
            if (window.cycleActivePage) window.cycleActivePage(); 
        } else if (e.key === 'Escape') {
            e.preventDefault();
            if (window.press) window.press({id: 'esc'});
        } else if (e.key.length === 1) { 
            // Bypass preventDefault for native OS clipboard shortcuts (Ctrl/Cmd + C, X, V, A)
            if (e.ctrlKey || e.metaKey) {
                let k = e.key.toLowerCase();
                if (['c', 'x', 'v', 'a'].includes(k)) return; 
            }
            window.press({char: e.key}); 
            e.preventDefault(); 
        }
        
    });

    document.addEventListener('keyup', function(e) {
        if (!window.setModifier) return;
        
        if (e.key === 'Shift') {
            window.setModifier('shift', 'physical-shift', false);
            window.setKeyVisual('lshift', 'active', false);
            window.setKeyVisual('rshift', 'active', false);
            window.updateKeyLabels();
        } else if (e.key === 'Control') {
            window.setModifier('ctrl', 'physical-ctrl', false);
            window.setKeyVisual('lctrl', 'active', false);
            window.setKeyVisual('rctrl', 'active', false);
            window.updateKeyLabels();
        } else if (e.key === 'Alt') {
            window.setModifier('alt', 'physical-alt', false);
            window.setKeyVisual('lalt', 'active', false);
            window.setKeyVisual('ralt', 'active', false);
            window.updateKeyLabels();
        }
    });
}

        async function init() {
            if (!window.parent || !window.parent.CylonDOS) {
                document.getElementById('error-msg').style.display = 'block';
                return;
            }
            let dos = window.parent.CylonDOS;
            await initThemes(dos);
            
            // "Everything is a file" - Read the active system variables directly from /a/
            if (!(await dos.exists('/a'))) await dos.mkdir('/a');
            
            for (let v of Object.keys(THEME_VARS)) {
                if (await dos.exists('/a/' + v)) {
                    let rawVal = await dos.load('/a/' + v);
                    // .trim() prevents newline breaks in CSS color assignments from Qandy dos.save operations
                    document.getElementById(v).value = rawVal ? rawVal.trim() : THEME_VARS[v];
                } else {
                    document.getElementById(v).value = THEME_VARS[v]; // Default fallback
                }
            }
            
            updateLivePreview();
        }

        // Strictly applies and activates the settings physically modifying the OS state
        async function applyTheme() {
            let btn = document.getElementById('apply-btn');
            if (!window.parent || !window.CylonScript) {
                if(btn) btn.textContent = "ERROR"; return;
            }

            let themeName = document.getElementById('themeName').value.replace(/[^a-zA-Z0-9_-]/g, '').trim() || 'default';
            document.getElementById('themeName').value = themeName; 
            
            let dos = window.parent.CylonDOS;
            if (!(await dos.exists('/a'))) await dos.mkdir('/a');

            // 1. Build a DOS Batch Script
            let script = "echo {.green}Applying Theme Variables...{.}\n";

            for (let v of Object.keys(THEME_VARS)) {
                let el = document.getElementById(v);
                let val = (v === 'themeName') ? themeName : (el ? el.value.trim() : THEME_VARS[v]);
                
                // Write each variable to its own file in /a/ using the OS CLI
                script += `save /a/${v} ${val}\n`;
            }

            try {
                // 2. Execute the batch script to physically save the variables
                await window.CylonScript(script);

                // 3. Force parent OS to hot-reload its CSS vars from the /a/ directory
                if (window.parent.applyThemeFromAlpha) {
                    await window.parent.applyThemeFromAlpha();
                }

                if (btn) {
                    let oldText = btn.textContent;
                    btn.textContent = "APPLIED!";
                    btn.style.backgroundColor = document.getElementById('themeForeground').value;
                    btn.style.color = document.getElementById('themeBackground').value;
                    setTimeout(() => { btn.textContent = oldText; updateLivePreview(); }, 1000);
                }
            } catch (err) {
                console.error("Failed to apply:", err);
                if(btn) { btn.textContent = "FAIL"; setTimeout(() => btn.textContent = "APPLY CHANGES", 1000); }
            }
        }

        window.onload = init;
    </script>
